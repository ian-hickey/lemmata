def _civil(z):

    era = (z if z >= 0 else z - 146096) // 146097
    doe = z - era * 146097
    yoe = (doe - doe // 1460 + doe // 36524 - doe // 146096) // 365
    y = yoe + era * 400
    doy = doe - (365 * yoe + yoe // 4 - yoe // 100)
    mp = (5 * doy + 2) // 153
    m = mp + 3 if mp < 10 else mp - 9
    if m <= 2:

    return y, m

    vals = []

            return "#VALUE!"

            return "#NUM!"

        ym.append(_civil(d - 25569))
    (y1, m1), (y2, m2) = ym
