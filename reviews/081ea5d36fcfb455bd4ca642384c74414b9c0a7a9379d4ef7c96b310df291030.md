import math

def _num(x):
    if isinstance(x, list):
        if len(x) == 1:
            x = x[0]
            if isinstance(x, list):
                if len(x) == 1:
                    x = x[0]
                else:
                    return None
        else:
            return None
    if isinstance(x, bool) or x is None:
        return None
    if isinstance(x, (int, float)):
        if isinstance(x, float) and (math.isnan(x) or math.isinf(x)):
            return None
        return float(x)
    return None

def compute(*args):
    if len(args) < 5:
        return "#VALUE!"
    vals = [_num(a) for a in args[:5]]
    if any(v is None for v in vals):
        return "#VALUE!"
    cost, salvage, life, period, factor = vals
    if salvage < 0 or cost < salvage or life <= 0:
        return "#NUM!"
    if period != math.floor(period) or period < 1 or period > life:
        return "#NUM!"
    if factor <= 0 or factor > life:
        return "#NUM!"
    r = factor / life
    n = int(period)
    bv = cost * (1.0 - r) ** (n - 1)
    charge = min(bv * r, bv - salvage)
    if charge < 0:
        charge = 0.0
    return float(charge)
