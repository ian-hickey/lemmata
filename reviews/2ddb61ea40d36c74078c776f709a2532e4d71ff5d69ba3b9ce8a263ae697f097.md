def _isnum(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool)

def compute(*args):

        return "#VALUE!"

        if not _isnum(a):
            return "#VALUE!"

            return "#NUM!"

        return "#NUM!"

        return "#NUM!"
    if periods <= 0 or periods != math.floor(periods):
        return "#NUM!"
    if period != math.floor(period) or period < 1 or period > periods:
        return "#NUM!"
    n = int(periods)
    k = int(period) - 1
    P = float(principal)
    r = float(rate)
    if r == 0:
        return 0.0
    try:
        lg = math.log1p(r)
        a = math.exp((k - n) * lg)
        b = -math.expm1(-n * lg)
        if b == 0:
            return "#NUM!"
        bal = P * (1.0 - a) / b
    except (OverflowError, ValueError, ZeroDivisionError):
        return "#NUM!"
    res = bal * r
    if math.isnan(res) or math.isinf(res):
        return "#NUM!"
    if res == 0:
        return 0.0
    return float(res)
