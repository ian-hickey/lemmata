import math

def _isnum(x):
    if isinstance(x, bool):
        return False
    if isinstance(x, (int, float)):
        return math.isfinite(float(x))
    return False

def _round15(v):
    if v == 0 or not math.isfinite(v):
        return v
    return float('%.15g' % v)

def compute(*args):
    if len(args) != 2:
        return '#VALUE!'
    revenue, cogs = args
    if not _isnum(revenue) or not _isnum(cogs):
        return '#VALUE!'
    revenue = float(revenue)
    cogs = float(cogs)
    if revenue <= 0 or cogs < 0:
        return '#NUM!'
    result = (revenue - cogs) / revenue
    return _round15(result)
