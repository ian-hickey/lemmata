Assumptions: Balance after k payments computed in the closed form P*(1-(1+r)^(k-n))/(1-(1+r)^-n), which equals the standard amortization balance; interest = balance after period-1 payments times rate, returned positive. import math

def _isnum(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool)

def compute(*args):
    if len(args) != 4:
        return "#VALUE!"
    for a in args:
        if not _isnum(a):
            return "#VALUE!"
        if isinstance(a, float) and (math.isnan(a) or math.isinf(a)):
            return "#NUM!"
    principal, rate, periods, period = [float(a) for a in args]
    if principal < 0:
        return "#NUM!"
    if rate < 0 or (rate != 0 and rate < 0.000001):
        return "#NUM!"
    if periods <= 0 or not periods.is_integer():
        return "#NUM!"
    if not period.is_integer() or period < 1 or period > periods:
        return "#NUM!"
    if rate == 0 or principal == 0:
        return 0.0
    n = periods
    k = period - 1
    lr = math.log1p(rate)
    # balance after k payments: P*(1-(1+r)^(k-n))/(1-(1+r)^(-n))
    num = -math.expm1((k - n) * lr)
    den = -math.expm1(-n * lr)
    if den == 0:
        return "#NUM!"
    bal = principal * num / den
    res = bal * rate
    if math.isnan(res) or math.isinf(res):
        return "#NUM!"
    return float(res)
