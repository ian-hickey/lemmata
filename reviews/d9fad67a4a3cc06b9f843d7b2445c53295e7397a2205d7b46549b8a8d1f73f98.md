def _isnum(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool)

def _round15(x):
    if x == 0 or not math.isfinite(x):
        return x
    d = 15 - int(math.floor(math.log10(abs(x)))) - 1
    return round(x, d)

def compute(*args):

        if not math.isfinite(a):
            return "#NUM!"
    if principal < 0:
        return "#NUM!"
    if rate < 0 or (rate != 0 and rate < 0.000001):
        return "#NUM!"
    if periods <= 0 or periods != math.floor(periods):
        return "#NUM!"
    if made < 0 or made != math.floor(made) or made > periods:
        return "#NUM!"
    n = int(periods)
    k = int(made)
    if k == n:
        return 0.0
    if rate == 0:
        bal = principal * (1 - k / n)
    else:
        try:
            payment = principal * rate / (1 - (1 + rate) ** (-n))
            bal = payment * (1 - (1 + rate) ** (-(n - k))) / rate
        except (OverflowError, ZeroDivisionError):
            return "#NUM!"
    if not math.isfinite(bal):
        return "#NUM!"
    return float(_round15(bal))
