Written from the cited definition and the parameter contract only. Assumptions: Booleans, text, and blank (None) arguments are not numbers and return #VALUE!. A 1x1 range is treated as its single value. The type check is done before the negative-tolerance #NUM! check. The absolute difference and the tolerance are both rounded to 15 significant digits before comparing. This mimics Excel's 15-digit precision, so for example ABS(1.1-1)<=0.1 gives TRUE. Non-finite inputs return #NUM!..

import math

def _round15(x):
    if x == 0 or not math.isfinite(x):
        return x
    d = 15 - int(math.floor(math.log10(abs(x)))) - 1
    try:
        return round(x, d)
    except Exception:
        return x

def _isnum(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool)

def compute(*args):
    if len(args) != 3:
        return "#VALUE!"
    vals = []
    for a in args:
        if isinstance(a, list) and len(a) == 1 and isinstance(a[0], list) and len(a[0]) == 1:
            a = a[0][0]
        if not _isnum(a):
            return "#VALUE!"
        a = float(a)
        if not math.isfinite(a):
            return "#NUM!"
        vals.append(a)
    value, target, tol = vals
    if tol < 0:
        return "#NUM!"
    diff = _round15(abs(value - target))
    return bool(diff <= _round15(tol))
