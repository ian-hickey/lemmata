import math

def _isnum(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool)

def compute(*args):
    if len(args) != 4:
        return "#VALUE!"
    principal, rate, periods, period = args
    for a in (principal, rate, periods, period):
        if not _isnum(a):
            return "#VALUE!"
    for a in (principal, rate, periods, period):
        if isinstance(a, float) and (math.isnan(a) or math.isinf(a)):
            return "#NUM!"
    if principal < 0:
        return "#NUM!"
    if rate < 0 or (rate != 0 and rate < 0.000001):
        return "#NUM!"
    if periods != math.floor(periods) or periods < 1:
        return "#NUM!"
    if period != math.floor(period) or period < 1 or period > periods:
        return "#NUM!"
    n = int(periods)
    k = int(period)
    P = float(principal)
    r = float(rate)
    if r == 0:
        res = P / n
    else:
        L = math.log1p(r)
        denom = -math.expm1(-n * L)
        res = P * r * math.exp((k - 1 - n) * L) / denom
    if res == 0:
        return 0.0
    try:
        res = float('%.15g' % res)
    except Exception:
        pass
    return float(res)
