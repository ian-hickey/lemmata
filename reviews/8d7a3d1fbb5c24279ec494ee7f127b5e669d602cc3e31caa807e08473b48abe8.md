Written from the cited definition and the parameter contract only. Assumptions: All #VALUE! checks (cells, expected_total, tolerance) precede the #NUM! negative-tolerance check. An empty values range is treated as sum 0. The sum and the absolute difference are rounded to 15 significant digits before comparison to mimic Excel precision (e.g. 0.1+0.2 vs 0.3 with tolerance 0 gives TRUE). Single-cell lists passed for scalar parameters are unwrapped..

```python
import math

def _isnum(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool)

def _r15(x):
    if x == 0 or not math.isfinite(x):
        return x
    d = 15 - int(math.floor(math.log10(abs(x)))) - 1
    try:
        return round(x, d)
    except Exception:
        return x

def _unwrap(x):
    while isinstance(x, (list, tuple)) and len(x) == 1:
        x = x[0]
    return x

def compute(*args):
    if len(args) < 3:
        return "#VALUE!"
    values, expected_total, tolerance = args[0], args[1], args[2]
    if not isinstance(values, (list, tuple)):
        values = [[values]]
    cells = []
    for row in values:
        if isinstance(row, (list, tuple)):
            cells.extend(row)
        else:
            cells.append(row)
    for c in cells:
        if not _isnum(c):
            return "#VALUE!"
    expected_total = _unwrap(expected_total)
    tolerance = _unwrap(tolerance)
    if not _isnum(expected_total) or not _isnum(tolerance):
        return "#VALUE!"
    if tolerance < 0:
        return "#NUM!"
    s = math.fsum(float(c) for c in cells)
    s = _r15(s)
    diff = _r15(abs(s - float(expected_total)))
    return bool(diff <= _r15(float(tolerance)))
```
